cmake_minimum_required(VERSION 3.24)
project(Onslaught LANGUAGES CXX RC)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "Use Windows + Visual Studio 2022 (MSVC) + Windows SDK 10.0.19041 ou superior")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_executable(OnslaughtEngine src/engine.cpp assets/app.rc)
target_compile_definitions(OnslaughtEngine PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(OnslaughtEngine PRIVATE /W4 /EHsc /utf-8)
target_link_libraries(OnslaughtEngine PRIVATE d3d11 dxgi d2d1 dwmapi windowsapp user32 gdi32 Magnification winmm)
add_executable(Onslaught WIN32 src/ui.cpp assets/app.rc)
target_compile_definitions(Onslaught PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(Onslaught PRIVATE /W4 /EHsc /utf-8)
target_link_libraries(Onslaught PRIVATE user32 gdi32 gdiplus dwmapi uxtheme comctl32 shell32 ole32 advapi32 winhttp)
add_executable(mapping_tests tests/mapping.cpp)
target_include_directories(mapping_tests PRIVATE src)
enable_testing()
add_test(NAME mapping COMMAND mapping_tests)
add_executable(cursor_tests tests/cursor.cpp)
target_include_directories(cursor_tests PRIVATE src)
target_compile_definitions(cursor_tests PRIVATE UNICODE _UNICODE NOMINMAX)
target_link_libraries(cursor_tests PRIVATE user32 gdi32)
add_test(NAME cursor COMMAND cursor_tests)

add_test(NAME interface COMMAND Onslaught --ui-self-test)

add_executable(release_tests tests/release.cpp)
target_include_directories(release_tests PRIVATE src)
add_test(NAME releases COMMAND release_tests)
add_executable(instance_tests tests/instance.cpp)
target_include_directories(instance_tests PRIVATE src)
target_compile_definitions(instance_tests PRIVATE UNICODE _UNICODE NOMINMAX)
target_link_libraries(instance_tests PRIVATE user32)
add_test(NAME instance COMMAND instance_tests)
